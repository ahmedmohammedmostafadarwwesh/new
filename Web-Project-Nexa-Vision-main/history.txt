    <script>
        // البيانات
        let historyData = [
            { name: "أحمد محمد", openTime: "09:30", closeTime: "17:15", exitFromOpenTime: "13:45", status: "مغلق" },
            { name: "سارة علي", openTime: "10:00", closeTime: "", exitFromOpenTime: "12:30", status: "خرج" },
            { name: "محمد خالد", openTime: "08:45", closeTime: "", exitFromOpenTime: "", status: "مفتوح" },
            { name: "نورا حسن", openTime: "14:00", closeTime: "16:30", exitFromOpenTime: "15:20", status: "مغلق" },
            { name: "كريم سمير", openTime: "11:20", closeTime: "", exitFromOpenTime: "11:50", status: "خرج" }
        ];

        function getStatusBadge(status) {
            if (status === "مفتوح") return '<span class="status-badge status-open">مفتوح</span>';
            if (status === "مغلق") return '<span class="status-badge status-closed">مغلق</span>';
            return '<span class="status-badge status-exited">خرج</span>';
        }

        function formatTime(time) {
            if (!time || time === "") return "—";
            return time;
        }

        function renderTimeline() {
            const container = document.getElementById("timelineContainer");
            
            container.innerHTML = historyData.map(item => `
                <div class="history-item">
                    <div class="info-row">
                        <div class="info-label">👤 الاسم:</div>
                        <div class="info-value"><strong>${item.name}</strong></div>
                    </div>
                    <div class="info-row">
                        <div class="info-label">📂 متى فتح:</div>
                        <div class="info-value">${formatTime(item.openTime)}</div>
                    </div>
                    <div class="info-row">
                        <div class="info-label">🔒 متى أغلق:</div>
                        <div class="info-value">${formatTime(item.closeTime)}</div>
                    </div>
                    <div class="info-row">
                        <div class="info-label">🚪 متى خرج من هذا فتح:</div>
                        <div class="info-value">${formatTime(item.exitFromOpenTime)}</div>
                    </div>
                    <div class="info-row">
                        <div class="info-label">📌 الحالة:</div>
                        <div class="info-value">${getStatusBadge(item.status)}</div>
                    </div>
                </div>
            `).join('');
        }

        // التنقل بين الصفحات
        function navigateTo(page) {
            let url = '';
            switch(page) {
                case 'home':
                    url = 'index.html';
                    break;
                case 'features':
                    url = 'index.html#features';
                    break;
                case 'tech':
                    url = 'index.html#tech';
                    break;
                case 'contact':
                    url = 'index.html#contact';
                    break;
                case 'data':
                    url = 'history.html';
                    break;
                case 'control':
                    url = 'Control.html';
                    break;
                default:
                    return;
            }
            window.location.href = url;
        }

        // إعدادات القائمة والتنقل
        const bar = document.getElementById("bar");
        const navbar = document.getElementById("navbar");

        if (bar) {
            bar.addEventListener("click", () => {
                navbar.classList.toggle("active");
            });
        }

        // إضافة حدث النقر على روابط القائمة
        document.querySelectorAll("#navbar li a").forEach(link => {
            link.addEventListener("click", (e) => {
                e.preventDefault();
                const page = link.getAttribute("data-page");
                navigateTo(page);
            });
        });

        // تشغيل عرض البيانات
        renderTimeline();
    </script>
